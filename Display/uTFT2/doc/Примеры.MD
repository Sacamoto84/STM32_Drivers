# Практические примеры использования библиотеки `uTFT2`

В данном руководстве собраны типовые примеры реализации интерфейсов на микроконтроллерах STM32 с использованием библиотеки **uTFT2**.

---

## 1. Базовая инициализация и отрисовка примитивов

Пример создания графического интерфейса со статус-баром, карточками и индикаторами:

```cpp
#include "main.h"
#include "TFT.h"

TFT tft;
uint16_t fb[135 * 240];

// Конфигурация дескриптора экрана ST7789
TFT_LCD_t LCD_0 = {
    .TFT_WIDTH = 135, .TFT_HEIGHT = 240,
    .LCD_Driver = ST7789, .Bit = 16,
    .hspi = &hspi1, .buffer16 = fb,
    .dx = 52, .dy = 40,
    .GPIO_CS = LCD_CS_GPIO_Port, .GPIO_Pin_CS = LCD_CS_Pin,
    .GPIO_DC = LCD_DC_GPIO_Port, .GPIO_Pin_DC = LCD_DC_Pin,
    .GPIO_RESET = LCD_RST_GPIO_Port, .GPIO_Pin_RESET = LCD_RST_Pin
};

void DrawUI(void) {
    tft.init(&LCD_0);

    // 1. Фон экрана
    tft.Fill(BLACK);

    // 2. Верхняя панель (Status Bar)
    tft.RectangleFilled(0, 0, 135, 24, DARKBLUE);
    tft.LineH(24, 0, 134, CYAN);

    // 3. Индикатор статуса (зеленый круг)
    tft.CircleFilled(120, 12, 5, GREEN);

    // 4. Информационная карточка
    tft.RectangleFilled(8, 35, 119, 70, DARKGREY);
    tft.Rectangle(8, 35, 119, 70, WHITE);

    // 5. Отправка буфера в контроллер дисплея
    tft.driver.ST7789_Update();
}
```

## 2. Вывод текста различными шрифтовыми движками

```cpp
#include "FontGFX/gfxfont.h"
#include "FontGFX/gfx_fonts_includes.h"
#include "FontSmooth/FontSmooth.h"
#include "FontClassic/FontClassic.h"

GFXFONT gfx;

void DrawTextDemo(void) {
    // 1. Классический моноширинный шрифт (для отладки/консоли)
    tft.SetFontColor(WHITE, BLACK);
    tft.GotoXY(10, 10);
    FontClassicPuts(&tft, (char*)"CPU: 168MHz\nRAM: OK", &Font_7x10, 0);

    // 2. GFX шрифт с кириллицей UTF-8
    gfx.init(&tft);
    gfx.setFont(&FreeSans9pt7b);
    tft.GotoXY(10, 60);
    tft.SetFontColor(YELLOW, BLACK);
    gfx.Puts("Температура:");

    // 3. Сглаженный шрифт Roboto 24pt
    Font_Smooth_Load(_acRoboto_Medium_en_ru_24);
    tft.SetFontColor(GREEN, BLACK);
    Font_Smooth_drawStr(&tft, 10, 90, "+24.5 °C");

    tft.driver.ST7789_Update();
}
```

## 3. Стрелочный прибор (вращение стрелки спрайтом)

Пример создания стрелочного индикатора (спидометра/вольтметра) с вращением спрайта стрелки вокруг оси:

```cpp
#include "Transformation/transformation.h"

// Создаем спрайт для стрелки прибора (ширина 5, длина 40 пикселей)
CREATE_SPRITE16(SprNeedle, 5, 40);

TFT needle;
transformation trans(&tft);

void InitNeedle(void) {
    needle.init(&SprNeedle);
    needle.Fill(BLACK); // Прозрачный цвет

    // Рисуем красную клиновидную стрелку
    needle.TriangleFilled(0, 40, 4, 40, 2, 0, RED);

    // Точка вращения внутри спрайта стрелки (ее основание)
    needle._xPivot = 2;
    needle._yPivot = 38;

    // Центр циферблата на основном экране
    tft._xPivot = 67;
    tft._yPivot = 140;
}

void UpdateGauge(int16_t angleDegrees) {
    // 1. Очищаем область циферблата
    tft.CircleFilled(67, 140, 45, BLACK);
    tft.Circle(67, 140, 45, WHITE);

    // 2. Рисуем повернутую стрелку
    trans.pushRotated(&tft, &needle, angleDegrees, BLACK);

    // 3. Центральная заглушка оси
    tft.CircleFilled(67, 140, 4, LIGHTGREY);

    tft.driver.ST7789_Update();
}
```

## 4. График реального времени с аппаратным скроллингом

Непрерывное построение графика АЦП или датчика с автоматической прокруткой влево:

```cpp
// Спрайт под область графика размером 120x60
CREATE_SPRITE16(SprGraph, 120, 60);

TFT graph;
transformation transGraph(&graph);

void InitGraph(void) {
    graph.init(&SprGraph);
    graph.Fill(BLACK);
    // Настраиваем зону прокрутки на весь спрайт
    transGraph.setScrollRect(0, 0, 120, 60, BLACK);
}

void AddGraphValue(uint16_t rawSensorValue) {
    // Масштабируем значение 0..4095 в высоту 0..59
    int32_t y = 59 - (rawSensorValue * 59 / 4095);

    // 1. Сдвигаем предыдущие значения графика на 1 пиксель влево
    transGraph.scroll(-1, 0);

    // 2. Рисуем новый столбик/точку на правом краю
    graph.SetPixel(119, y, GREEN);

    // 3. Копируем спрайт графика в область экрана (X=8, Y=80)
    tft.copy(&graph, 8, 80);
    tft.Rectangle(7, 79, 122, 62, DARKGREY); // Рамка вокруг графика

    tft.driver.ST7789_Update();
}
```

## 5. Воспроизведение анимированных GIF с карты памяти

При включенном модуле `#define FAT_FS 1`:

```cpp
#include "Animated/Gif.h"

Gif gifPlayer;

void PlayAnimation(void) {
    gifPlayer.init(&tft);
    gifPlayer.setName((char*)"cat_anim"); // Папка /Gif/cat_anim/ на SD-карте

    while (!gifPlayer.isFinished()) {
        gifPlayer.calculate(); // Воспроизводит кадры с учетом пауз
    }
}
```
